import React, { useState } from 'react';
import { 
  ShoppingBag, 
  User, 
  Search, 
  Menu, 
  X, 
  Compass, 
  Home, 
  Heart, 
  ChevronRight 
} from 'lucide-react';

export default function MyntraNavbar() {
  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false);
  const [activeTab, setActiveTab] = useState('Home');
  const [cartCount, setCartCount] = useState(3);

  const navLinks = [
    { name: 'Home', icon: Home, href: '#' },
    { name: 'Explore', icon: Compass, href: '#' },
    { name: 'Categories', icon: null, href: '#', isCategory: true },
    { name: 'Studio', icon: null, href: '#', badge: 'NEW' },
  ];

  return (
    <div class="w-full relative font-sans">
      {/* =====================================================
          DESKTOP & MOBILE HEADER (STICKY TOP)
         ===================================================== */}
      <header class="sticky top-0 z-50 bg-white shadow-sm border-b border-gray-200">
        <div class="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 h-20 flex items-center justify-between gap-4">
          
          {/* Logo & Mobile Menu Toggle */}
          <div class="flex items-center gap-3">
            <button 
              onClick={() => setIsMobileMenuOpen(!isMobileMenuOpen)}
              class="lg:hidden p-2 text-gray-600 hover:text-[#ff3f6c] focus:outline-none"
              aria-label="Toggle menu"
            >
              {isMobileMenuOpen ? <X size={24} /> : <Menu size={24} />}
            </button>

            {/* Myntra Logo */}
            <a href="#" class="flex items-center gap-2 group">
              <div class="w-10 h-10 rounded-lg bg-gradient-to-tr from-[#ff3f6c] to-[#ff905a] flex items-center justify-center text-white font-extrabold text-xl shadow-md group-hover:scale-105 transition-transform">
                M
              </div>
              <span class="font-extrabold text-xl tracking-tight text-gray-900 hidden sm:inline-block">
                MYNTRA
              </span>
            </a>
          </div>

          {/* DESKTOP NAVIGATION LINKS */}
          <nav class="hidden lg:flex items-center space-x-8 h-full">
            {navLinks.map((link) => (
              <a
                key={link.name}
                href={link.href}
                onClick={() => setActiveTab(link.name)}
                class={`relative h-full flex items-center text-xs font-bold tracking-wider uppercase transition-colors px-1 ${
                  activeTab === link.name 
                    ? 'text-[#ff3f6c]' 
                    : 'text-gray-800 hover:text-[#ff3f6c]'
                }`}
              >
                {link.name}
                {link.badge && (
                  <span class="ml-1 -mt-3 text-[9px] bg-[#ff3f6c] text-white px-1 rounded font-bold">
                    {link.badge}
                  </span>
                )}
                {/* Active Underline Indicator */}
                {activeTab === link.name && (
                  <span class="absolute bottom-0 left-0 w-full h-[3px] bg-[#ff3f6c] rounded-t-md" />
                )}
              </a>
            ))}
          </nav>

          {/* DESKTOP SEARCH BAR */}
          <div class="flex-1 max-w-md hidden sm:block relative">
            <div class="relative flex items-center">
              <Search class="absolute left-3 text-gray-400 w-4 h-4" />
              <input
                type="text"
                placeholder="Search for products, brands and more"
                class="w-full bg-gray-100 text-xs text-gray-800 pl-10 pr-4 py-2.5 rounded-md focus:outline-none focus:bg-white focus:ring-1 focus:ring-[#ff3f6c] border border-transparent focus:border-[#ff3f6c] transition-all"
              />
            </div>
          </div>

          {/* ACTION BUTTONS (Profile, Wishlist, Cart) */}
          <div class="flex items-center space-x-6 text-xs font-semibold">
            
            {/* Profile */}
            <a 
              href="#" 
              onClick={() => setActiveTab('Profile')}
              class="flex flex-col items-center text-gray-700 hover:text-[#ff3f6c] transition-colors"
            >
              <User class="w-5 h-5 mb-0.5" />
              <span class="hidden md:inline">Profile</span>
            </a>

            {/* Wishlist */}
            <a 
              href="#" 
              class="hidden md:flex flex-col items-center text-gray-700 hover:text-[#ff3f6c] transition-colors"
            >
              <Heart class="w-5 h-5 mb-0.5" />
              <span>Wishlist</span>
            </a>

            {/* Cart with Badge */}
            <a 
              href="#" 
              onClick={() => setActiveTab('Cart')}
              class="flex flex-col items-center text-gray-700 hover:text-[#ff3f6c] transition-colors relative"
            >
              <div class="relative">
                <ShoppingBag class="w-5 h-5 mb-0.5" />
                {cartCount > 0 && (
                  <span class="absolute -top-1.5 -right-2 bg-[#ff3f6c] text-white text-[10px] font-extrabold w-4 h-4 rounded-full flex items-center justify-center shadow-sm">
                    {cartCount}
                  </span>
                )}
              </div>
              <span class="hidden md:inline">Bag</span>
            </a>
          </div>
        </div>

        {/* MOBILE SEARCH BAR BELOW HEADER */}
        <div class="p-2 bg-gray-50 border-t border-gray-200 sm:hidden">
          <div class="relative flex items-center">
            <Search class="absolute left-3 text-gray-400 w-4 h-4" />
            <input
              type="text"
              placeholder="Search for products, brands..."
              class="w-full bg-white text-xs text-gray-800 pl-9 pr-3 py-2 rounded-md border border-gray-300 focus:outline-none focus:border-[#ff3f6c]"
            />
          </div>
        </div>
      </header>

      {/* =====================================================
          MOBILE DRAWER MENU (SLIDE OUT)
         ===================================================== */}
      {isMobileMenuOpen && (
        <div class="fixed inset-0 z-50 flex lg:hidden">
          {/* Backdrop */}
          <div 
            class="fixed inset-0 bg-black/50 backdrop-blur-sm" 
            onClick={() => setIsMobileMenuOpen(false)}
          />

          {/* Drawer Panel */}
          <div class="relative w-4/5 max-w-sm bg-white h-full shadow-2xl flex flex-col z-10">
            {/* Drawer Header */}
            <div class="p-4 bg-gradient-to-r from-[#ff3f6c] to-[#ff905a] text-white flex items-center justify-between">
              <div class="flex items-center gap-3">
                <div class="w-10 h-10 rounded-full bg-white text-[#ff3f6c] font-bold flex items-center justify-center">
                  U
                </div>
                <div>
                  <p class="font-bold text-sm">Welcome Guest</p>
                  <p class="text-xs text-pink-100">To access orders & wishlist</p>
                </div>
              </div>
              <button onClick={() => setIsMobileMenuOpen(false)} class="text-white hover:text-gray-200">
                <X size={20} />
              </button>
            </div>

            {/* Menu Items */}
            <div class="flex-1 overflow-y-auto p-4 space-y-1 text-sm font-semibold text-gray-700">
              <a href="#" class="flex items-center justify-between py-3 border-b hover:text-[#ff3f6c]">
                <span>Men Fashion</span> <ChevronRight size={16} />
              </a>
              <a href="#" class="flex items-center justify-between py-3 border-b hover:text-[#ff3f6c]">
                <span>Women Fashion</span> <ChevronRight size={16} />
              </a>
              <a href="#" class="flex items-center justify-between py-3 border-b hover:text-[#ff3f6c]">
                <span>Kids & Baby</span> <ChevronRight size={16} />
              </a>
              <a href="#" class="flex items-center justify-between py-3 border-b hover:text-[#ff3f6c]">
                <span>Home & Living</span> <ChevronRight size={16} />
              </a>
              <a href="#" class="flex items-center justify-between py-3 border-b hover:text-[#ff3f6c]">
                <span>Beauty & Personal Care</span> <ChevronRight size={16} />
              </a>
              <a href="#" class="flex items-center justify-between py-3 hover:text-[#ff3f6c]">
                <span>Myntra Studio</span> <span class="text-xs text-[#ff3f6c] font-bold uppercase">New</span>
              </a>
            </div>
          </div>
        </div>
      )}

      {/* =====================================================
          MOBILE BOTTOM NAVIGATION BAR
         ===================================================== */}
      <div class="fixed bottom-0 left-0 right-0 bg-white border-t border-gray-200 z-40 lg:hidden flex justify-around items-center py-2 shadow-lg">
        <button 
          onClick={() => setActiveTab('Home')}
          class={`flex flex-col items-center text-[10px] font-bold ${
            activeTab === 'Home' ? 'text-[#ff3f6c]' : 'text-gray-500'
          }`}
        >
          <Home size={20} class="mb-0.5" />
          Home
        </button>

        <button 
          onClick={() => setActiveTab('Explore')}
          class={`flex flex-col items-center text-[10px] font-bold ${
            activeTab === 'Explore' ? 'text-[#ff3f6c]' : 'text-gray-500'
          }`}
        >
          <Compass size={20} class="mb-0.5" />
          Explore
        </button>

        <button 
          onClick={() => setActiveTab('Cart')}
          class={`flex flex-col items-center text-[10px] font-bold relative ${
            activeTab === 'Cart' ? 'text-[#ff3f6c]' : 'text-gray-500'
          }`}
        >
          <ShoppingBag size={20} class="mb-0.5" />
          {cartCount > 0 && (
            <span class="absolute -top-1 right-2 bg-[#ff3f6c] text-white text-[9px] w-3.5 h-3.5 rounded-full flex items-center justify-center font-bold">
              {cartCount}
            </span>
          )}
          Cart
        </button>

        <button 
          onClick={() => setActiveTab('Profile')}
          class={`flex flex-col items-center text-[10px] font-bold ${
            activeTab === 'Profile' ? 'text-[#ff3f6c]' : 'text-gray-500'
          }`}
        >
          <User size={20} class="mb-0.5" />
          Profile
        </button>
      </div>

    </div>
  );
}
